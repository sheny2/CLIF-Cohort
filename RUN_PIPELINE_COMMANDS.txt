# Hopkins CLIF severe-CAP target-trial cohort workflow
# Run on the Discovery HPC. Patient-level outputs must remain on Discovery.

# 1. Enter the project directory.
cd /projects/emulate_icu/CLIF-Cohort

# 2. Define source and output locations.
# Shell variable assignments cannot contain spaces around the equals sign.
export CLIF_DATA_DIR=/projects/emulate_icu/rclif_2_1
export PIPELINE_OUTPUT_ROOT=./pipeline_output
export SENSITIVE_ROOT="$PIPELINE_OUTPUT_ROOT/sensitive"
export SHAREABLE_ROOT="$PIPELINE_OUTPUT_ROOT/shareable"

# Patient-level and potentially disclosive intermediate data stay here.
export INPUT_CHECK_DIR="$SENSITIVE_ROOT/input_checks"
export GLUCOCORTICOID_CHECK_DIR="$SENSITIVE_ROOT/glucocorticoid_source_profile"
export COHORT_OUTPUT_DIR="$SENSITIVE_ROOT/cohort"
export STEROID_CANDIDATE_DIR="$SENSITIVE_ROOT/steroid_candidates"
export STEROID_TREATMENT_DIR="$SENSITIVE_ROOT/steroid_treatment"
export CCW_SENSITIVE_DIR="$SENSITIVE_ROOT/ccw_analysis"

# Disclosure-controlled summaries and figures that may be reviewed for export.
export SUMMARY_EDA_DIR="$SHAREABLE_ROOT/cohort_summary_eda"
export CCW_SHAREABLE_DIR="$SHAREABLE_ROOT/ccw_results"

mkdir -p "$SENSITIVE_ROOT" "$SHAREABLE_ROOT"

# 3. Reproducibility parameters.
export PNEUMONIA_CASE_DEFINITION=poa_any
export HOSPITAL_BLOCK_GAP_HOURS=6
export COHORT_SELECTION_SEED=32284
export CLIF_SITE_ID=hopkins_unspecified
export TREATMENT_GRACE_HOURS=24
export SUMMARY_SMALL_CELL_THRESHOLD=10
export CLIF_DATA_CUTOFF_DATE=2026-10-31
export CCW_MAX_PATIENTS=0

# 4. Validate files, schemas, category values, POA, lactate, and pressors.
Rscript 00_check_clif_cohort_inputs.R \
  "$CLIF_DATA_DIR" \
  "$INPUT_CHECK_DIR"

# 5. Build the initiation-based severe-CAP cohort.
Rscript 01_build_adult_pneumonia_cohort.R \
  "$CLIF_DATA_DIR" \
  "$COHORT_OUTPUT_DIR" \
  "$PNEUMONIA_CASE_DEFINITION" \
  "$HOSPITAL_BLOCK_GAP_HOURS" \
  "$COHORT_SELECTION_SEED" \
  "$CLIF_SITE_ID"

# 6. Profile glucocorticoid source fields and categories.
Rscript 02_profile_glucocorticoid_sources.R \
  "$CLIF_DATA_DIR" \
  "$GLUCOCORTICOID_CHECK_DIR"

# 7. Extract systemic steroid candidates linked to the final cohort.
Rscript 03_extract_candidate_steroid_administrations.R \
  "$CLIF_DATA_DIR" \
  "$COHORT_OUTPUT_DIR" \
  "$STEROID_CANDIDATE_DIR"

# 8. Apply drug-, dose-, route-, and 24-hour grace-period rules.
Rscript 04_define_steroid_treatment.R \
  "$STEROID_CANDIDATE_DIR" \
  "$COHORT_OUTPUT_DIR" \
  "$STEROID_TREATMENT_DIR" \
  "$TREATMENT_GRACE_HOURS"

# 9. Produce disclosure-controlled aggregate summaries and figures.
# Use the prespecified October 31, 2026 follow-up cutoff.
Rscript 05_cohort_summary_eda.R \
  "$CLIF_DATA_DIR" \
  "$COHORT_OUTPUT_DIR" \
  "$STEROID_TREATMENT_DIR" \
  "$SUMMARY_EDA_DIR" \
  "$SUMMARY_SMALL_CELL_THRESHOLD" \
  "$CLIF_DATA_CUTOFF_DATE"

# 10. Run the five-method target-trial analysis. Script 06 contains the method
# library and is sourced automatically by script 07.
Rscript 07_prepare_and_run_five_methods.R \
  "$CLIF_DATA_DIR" \
  "$COHORT_OUTPUT_DIR" \
  "$STEROID_TREATMENT_DIR" \
  "$CCW_SENSITIVE_DIR" \
  "$CCW_SHAREABLE_DIR" \
  "$CCW_MAX_PATIENTS" \
  "$CLIF_DATA_CUTOFF_DATE"

# 11. Review aggregate outputs that may be moved only after disclosure review.
# Everything under sensitive/ must remain on the approved Discovery HPC.
# Files under shareable/ are aggregate and disclosure-controlled, but still
# require the project's usual disclosure review before being moved.
ls -lh "$INPUT_CHECK_DIR"
ls -lh "$SUMMARY_EDA_DIR"
ls -lh "$CCW_SHAREABLE_DIR"

# Standalone analysis dated 2026-11-24. Run in bash on server terminal.
# Change the first cd if project is stored elsewhere.
set -euo pipefail

cd /projects/emulate_icu/CLIF-Cohort/Analysis_2026_11_24


# Relative to the CLIF-Cohort folder, source data remain ../rclif_2_1.
# This analysis folder is directly inside CLIF-Cohort.
export CLIF_DATA_DIR="$(cd ../.. && pwd)/rclif_2_1"
# Limit native data.table parallelism; 01 also enforces this explicitly.
export R_DATATABLE_NUM_THREADS=1
export TREATMENT_GRACE_HOURS="${TREATMENT_GRACE_HOURS:-24}"
# Analysis interval length; must divide both grace and 672 hours exactly.
export HOSPITAL_BLOCK_HOURS="${HOSPITAL_BLOCK_HOURS:-6}"
export CLIF_DATA_CUTOFF_DATE=2026-10-31
export HOSPITAL_BLOCK_GAP_HOURS="${HOSPITAL_BLOCK_GAP_HOURS:-6}"  # encounter-merging gap; separate from analysis intervals
export COHORT_SELECTION_SEED=2026
export PNEUMONIA_CASE_DEFINITION=poa_any  # qualifying pneumonia diagnosis present on admission
export CLIF_SITE_ID=hopkins_unspecified
export SUMMARY_SMALL_CELL_THRESHOLD=1 


# Validate timing before running any cohort-building work.
Rscript -e 'source("analysis_preparation.R"); analysis_timing(as.numeric(Sys.getenv("TREATMENT_GRACE_HOURS")))'

# A fresh result directory for every run
export RUN_ROOT="$PWD/results/run_$(date +%Y%m%d_%H%M%S)_grace${TREATMENT_GRACE_HOURS}h_block${HOSPITAL_BLOCK_HOURS}h_gap${HOSPITAL_BLOCK_GAP_HOURS}h"
export COHORT_OUTPUT_DIR="$RUN_ROOT/sensitive/cohort"
export GLUCOCORTICOID_CHECK_DIR="$RUN_ROOT/sensitive/source_profile"
export STEROID_CANDIDATE_DIR="$RUN_ROOT/sensitive/steroid_candidates"
export STEROID_TREATMENT_DIR="$RUN_ROOT/sensitive/steroid_treatment"
export SUMMARY_EDA_DIR="$RUN_ROOT/aggregate/eda"
mkdir -p "$RUN_ROOT/logs"


Rscript 01_build_adult_pneumonia_cohort.R \
 "$CLIF_DATA_DIR" "$COHORT_OUTPUT_DIR" "$PNEUMONIA_CASE_DEFINITION" \
 "$HOSPITAL_BLOCK_GAP_HOURS" "$COHORT_SELECTION_SEED" "$CLIF_SITE_ID" \
 2>&1 | tee "$RUN_ROOT/logs/01.log"


Rscript 02_profile_glucocorticoid_sources.R \
 "$CLIF_DATA_DIR" "$GLUCOCORTICOID_CHECK_DIR" \
 2>&1 | tee "$RUN_ROOT/logs/02.log"


Rscript 03_extract_candidate_steroid_administrations.R \
 "$CLIF_DATA_DIR" "$COHORT_OUTPUT_DIR" "$STEROID_CANDIDATE_DIR" \
 2>&1 | tee "$RUN_ROOT/logs/03.log"


Rscript 04_define_steroid_treatment.R \
 "$STEROID_CANDIDATE_DIR" "$COHORT_OUTPUT_DIR" "$STEROID_TREATMENT_DIR" \
 "$TREATMENT_GRACE_HOURS" 2>&1 | tee "$RUN_ROOT/logs/04.log"


Rscript 05_cohort_summary_eda.R \
 "$CLIF_DATA_DIR" "$COHORT_OUTPUT_DIR" "$STEROID_TREATMENT_DIR" \
 "$SUMMARY_EDA_DIR" "$SUMMARY_SMALL_CELL_THRESHOLD" "$CLIF_DATA_CUTOFF_DATE" \
 "$RUN_ROOT/sensitive/eda_preparation" 2>&1 | tee "$RUN_ROOT/logs/05.log"


# 06 is a local method library, sourced automatically
Rscript 07_prepare_and_run_five_methods.R \
 "$CLIF_DATA_DIR" "$COHORT_OUTPUT_DIR" "$STEROID_TREATMENT_DIR" \
 "$RUN_ROOT" "$CLIF_DATA_CUTOFF_DATE" \
 2>&1 | tee "$RUN_ROOT/logs/07.log"


# Main summaries: $RUN_ROOT/aggregate/analysis/five_method_results.csv
# Weight diagnostics: $RUN_ROOT/aggregate/analysis/
# EDA including screening ranges: $SUMMARY_EDA_DIR/
# Individual records: sensitive/; retain on the data server.
# Aggregate counts/plots are unsuppressed, including cells below 10.
# Examples (or edit the defaults above):
# HOSPITAL_BLOCK_HOURS=3 bash RUN_PIPELINE_COMMANDS.txt
# HOSPITAL_BLOCK_HOURS=12 bash RUN_PIPELINE_COMMANDS.txt
# HOSPITAL_BLOCK_GAP_HOURS=12 bash RUN_PIPELINE_COMMANDS.txt
# TREATMENT_GRACE_HOURS=96 HOSPITAL_BLOCK_HOURS=12 bash RUN_PIPELINE_COMMANDS.txt
# The cohort/timing EDA continues to show both 24-hour and 96-hour comparisons.


# 08: independent site-local CCW + random-effects meta-analysis; reads completed 07.
# Extra package required: install.packages("metafor") in the R environment.
Rscript 08_local_ccw_meta_analysis.R "$RUN_ROOT" \
 2>&1 | tee "$RUN_ROOT/logs/08.log"
